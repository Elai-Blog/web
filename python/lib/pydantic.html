pydantic ​

Pydantic 是 Python 数据校验、类型转换库，基于类型注解，自动校验、自动类型转换，广泛用于 FastAPI、配置解析、接口入参校验、ORM映射。

安装 ​

bash
# v2版本
uv add pydantic

使用 ​

python
from datetime import datetime
from pydantic import BaseModel, ValidationError

class User(BaseModel):
    id: int               # 必填，int，自动转（字符串"123"→123）
    name: str = "默认名字" # 有默认值，非必填
    signup_time: datetime | None  # 可以是 datetime 或者 None，必填（没有默认）

# 原始数据（字典，甚至可以是JSON字符串）
raw_data = {"id": "1001", "signup_time": "2025-05-01 10:30:00"}

# 方式1：直接实例化
user = User(**raw_data)
print(user.id)
print(user.signup_time)

# 方式2：model_validate 推荐（替代v1 parse_obj）
user2 = User.model_validate(raw_data)

# 转字典
print(user.model_dump())
# 转json字符串
print(user.model_dump_json(indent=2))

# 校验失败捕获 ValidationError
try:
    User(id="abc", signup_time="2025-05-01")
except ValidationError as e:
    print(e)
    print(e.errors())  # 错误详情列表

Field ​

  • ge / gt / le / lt：大于等于、大于、小于等于、小于
  • min_length / max_length：字符串长度
  • pattern：正则
  • alias：别名（入参、序列化都生效）
  • validation_alias：只解析入参别名
  • serialization_alias：只序列化输出别名
  • default_factory：动态默认值（不能和 default 一起用，适合 list、dict、uuid）
  • exclude：序列化时排除此字段
  • description：文档说明
python
from uuid import uuid4
from pydantic import BaseModel, Field, EmailStr

class Product(BaseModel):
    id: int = Field(ge=1, description="商品ID，必须大于0")
    name: str = Field(min_length=2, max_length=50)
    price: float = Field(gt=0)
    email: EmailStr  # 内置邮箱校验
    tag_list: list[str] = Field(default_factory=list) # 每次新建独立空列表
    secret: str = Field(exclude=True) # 序列化时隐藏

    # 别名示例：外部传 product_id，内部字段是id
    pid: int = Field(alias="product_id")

data = {
    "id": 1,
    "name": "苹果",
    "price": 9.9,
    "email": "test@xx.com",
    "secret": "xxx",
    "product_id": 100
}
p = Product.model_validate(data)
print(p.model_dump(by_alias=True)) # by_alias=True 使用别名作为key

模型配置 ​

python
from pydantic import BaseModel, ConfigDict

class User(BaseModel):
    id: int
    name: str

    model_config = ConfigDict(
        extra="forbid",        # forbid：不允许多余字段；allow：保留多余字段；ignore：丢弃多余字段
        strict=True,           # 严格模式：不自动类型转换，"123"不能转int
        validate_assignment=True, # 修改属性时也校验 user.id="abc" 会报错
        from_attributes=True,  # ORM模式，支持从ORM对象读取属性（替代v1 orm_mode）
        frozen=True,           # 不可变模型，实例创建后不能修改
        use_enum_values=True,  # 序列化Enum输出值而不是枚举对象
        str_strip_whitespace=True, # 自动去除字符串首尾空格
    )

class Settings(BaseSettings):

Annotated ​

  • 替代Field校检
  • 支持叠加多个校验器 / 约束（Field + AfterValidator + WithJsonSchema 放一起）
python
# 传统写法（赋值式 Field）
age: int = Field(gt=0, description="年龄")

# Annotated写法（推荐，V2官方主推）
age: Annotated[int, Field(gt=0, description="年龄")]

默认值 ​

Annotated 内部的 Field 不能写 default=xxx！

python
from uuid import uuid4

def default_value():
    return lambda: uuid4().hex

class User(BaseModel):
    # 正确：default_factory 放内部
    uid: Annotated[str, Field(default_factory=default_value)]

    # 正确：默认值写外面
    name: Annotated[str, Field(min_length=2)] = "匿名用户"
    # ❌ 错误：Field(default="xxx") 写在 Annotated 里面不支持

可以复用 ​

python
from typing import Annotated
from pydantic import Field, BaseModel

PositiveInt = Annotated[int, Field(gt=0)]
UserNameStr = Annotated[str, Field(min_length=2, max_length=30)]

class Order(BaseModel):
    order_id: PositiveInt
    buyer_name: UserNameStr

class Goods(BaseModel):
    goods_id: PositiveInt
    title: UserNameStr

功能性校验器 ​

python
from typing import Annotated
from pydantic import BaseModel, AfterValidator, BeforeValidator, ValidationInfo

# ========== AfterValidator：pydantic基础校验完成之后执行（最常用）
def even_check(v: int):
    if v % 2 != 0:
        raise ValueError("必须是偶数")
    return v

EvenInt = Annotated[int, AfterValidator(even_check)]

# ========== BeforeValidator：在pydantic类型转换、校验之前执行
def str_to_int(v):
    # 先把字符串转成int，再交给pydantic校验
    if isinstance(v, str):
        return int(v)
    return v

StrToInt = Annotated[int, BeforeValidator(str_to_int)]

def wrap_func(v: Any, handler):
    # handler 是pydantic原生校验函数
    print("校验前", v)
    res = handler(v)
    print("校验后", res)
    return res

WrapInt = Annotated[int, WrapValidator(wrap_func)]

class Demo(BaseModel):
    num: EvenInt
    num2: StrToInt
    num3: WrapInt

print(Demo(num=4, num2="100",num3=50))

组合校检 ​

python
from typing import Annotated
from pydantic import BaseModel, Field, AfterValidator, WithJsonSchema

def check_gt10(v: int):
    if v <= 10:
        raise ValueError("大于10")
    return v

MyInt = Annotated[
    int,
    Field(description="自定义数字"),
    AfterValidator(check_gt10),
    WithJsonSchema({"examples": [20, 30]})
]

class Test(BaseModel):
    val: MyInt

print(Test.model_json_schema())
