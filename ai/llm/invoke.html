from rich import print

invoke（最常用） ​

一次性返回完整结果问答、批处理任务、无需实时反馈的场景。

python
messages = [
	{"role": "system", "content": "你是一个作家"},
	{"role": "user", "content": "我帮写首关于春天的诗"}
]
# 阻塞调用
def invoke_func():
	resp1 = llm.invoke(messages)
	print(resp1.content)

stream() ​

流式输出，实时返回每个token.聊天机器人、长文本生成、需要提升用户体验的交互应用。

python
messages = [
	{"role": "system", "content": "你是一个作家"},
	{"role": "user", "content": "我帮写首关于春天的诗"}
]
# 流式输出
def stream_func():
	for s in llm.stream(messages):
		print(s.content, end="", flush=True)

if __name__ == "__main__":
	stream_func()

batch() ​

批量处理多个输入高并发场景，需要同时处理大量请求。

python
messages = [
	{"role": "system", "content": "你是一个作家"},
	{"role": "user", "content": "我帮写首关于春天的诗"}
]
# batch 批量调用
def batch_func():
	batch_messages = [
		"你是谁？",
		"1+1=?",
		"请用一句话介绍自己！"
	]
	responses = llm.batch(batch_messages)
	print(responses)
	for r in responses:
		print(r)
if __name__ == "__main__":
	asyncio.run(ainvoke_func())

ainvoke() ​

异步调用，无需等待

python
def build_messages(user_content: str):
	return [
		{"role": "system", "content": "你是一个作家"},
		{"role": "user", "content": user_content},
	]

async def async_task(name: str, user_content: str):
	"""单个异步协程：发起请求后 await 会让出事件循环，切给其它任务"""
	start = time.perf_counter()
	print(f"[cyan]{name}[/cyan] 发起请求 t={start:.2f}")
	resp = await llm.ainvoke(build_messages(user_content))
	cost = time.perf_counter() - start
	print(f"[cyan]{name}[/cyan] 收到响应，耗时 {cost:.2f}s，共 {len(resp.content)} 字")
	return resp.content

async def serial_func():
	"""串行 await：每次 await 都在干等，事件循环里没有其它任务可切"""
	start = time.perf_counter()
	r1 = await async_task("任务1", "写一首关于春天的诗")
	r2 = await async_task("任务2", "写一首关于秋天的诗")
	print(f"[red]串行总耗时 {time.perf_counter() - start:.2f}s[/red] ≈ 任务1 + 任务2")
	return r1, r2

async def ainvoke_func():
	"""并发 gather：两个请求同时在飞，总耗时取决于最慢的那个"""
	start = time.perf_counter()
	r1, r2 = await asyncio.gather(
		async_task("任务1", "写一首关于春天的诗"),
		async_task("任务2", "写一首关于秋天的诗"),
	)
	print(f"[green]并发总耗时 {time.perf_counter() - start:.2f}s[/green] ≈ max(任务1, 任务2)")
	return r1, r2

if __name__ == "__main__":
	asyncio.run(ainvoke_func())
