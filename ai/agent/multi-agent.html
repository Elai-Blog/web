交接 ​

python
import config  # 加载设置
import asyncio
from agents import Agent, Runner, function_tool

agent_a = Agent(
	name="agent_a",
	instructions="只使用中文进行问答。",
	handoff_description="负责中文问题回答的助手",
)
agent_b = Agent(
	name="agent_b",
	instructions="只使用英文进行问答。",
	handoff_description="负责英文问题回答的助手",
)

agent_c = Agent(
	name="agent_c",
	instructions="只使用韩文进行问答。",
	handoff_description="负责韩文问题回答的助手",
)

agent_all = Agent(
	name="agent_all",
	instructions="你不回答问题，你把问题交接合适的agent",  # 系统提示词
	handoff_description="负责英文问题回答的助手",
	handoffs=[agent_a, agent_b, agent_c]  # 将对话委派给专业智能体。
)

async def run_llm_handoff():
	query = "太阳是围绕地球转，还是地球围绕太阳转？"
	resp = await  Runner.run(agent_all, query)
	print("最终回答：", resp.final_output)

if __name__ == "__main__":
	# 等待异步执行完毕
	asyncio.run(run_llm_handoff())
