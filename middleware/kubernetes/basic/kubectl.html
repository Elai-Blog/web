kubectl ​

  • 角色：客户端 CLI
  • 运行位置：所有 Node（master+worker）
  • 核心用途：增删改查集群资源
  • 是否日常操作：日常高频使用
bash
# 查看所有命令
kubectl --help

查看资源 ​

bash
# 查看所有节点
kubectl get nodes
kubectl get node -o wide

# 查看所有 Pod
kubectl get pods
kubectl get pod -o wide      # 详细信息（节点、IP）
kubectl get pod -n 命名空间  # 指定命名空间


# 查看 Deployment（无状态应用）
kubectl get deploy
# 查看 Service
kubectl get svc
# 查看 ConfigMap/Secret
kubectl get cm
kubectl get secret
# 查看所有资源
kubectl get all

详情描述 ​

bash
kubectl describe pod <pod名>
kubectl describe deploy <deploy名>
kubectl describe node <节点名>

查看日志 ​

bash
kubectl logs <pod名>
kubectl logs -f <pod名>           # 实时滚动日志
kubectl logs --previous <pod名>   # 查看上一次崩溃容器日志
kubectl logs <pod名> -c 容器名    # 多容器Pod指定容器日志

部署应用 ​

bash
# 命令行快速创建deployment
kubectl create deploy nginx --image=nginx

# 应用yaml配置（创建/更新通用）
kubectl apply -f xxx.yaml
# 批量应用目录下所有yaml
kubectl apply -f ./dir/

# 创建service暴露端口
kubectl expose deploy nginx --port=80 --type=NodePort

删除资源 ​

bash
kubectl delete pod <pod名>
kubectl delete deploy <deploy名>
kubectl delete svc <svc名>

# 删除指定yaml对应的资源
kubectl delete -f xxx.yaml

# 删除某个命名空间下所有pod
kubectl delete pods --all -n test

进入容器内部操作 ​

bash
# 交互式进入pod
kubectl exec -it <pod名> -- /bin/bash
# 容器只有sh
kubectl exec -it <pod名> -- sh
# 多容器指定容器进入
kubectl exec -it <pod名> -c 容器名 -- bash

文件拷贝 ​

bash
# 宿主机文件传到pod
kubectl cp /本地路径 pod名:/容器路径

# pod文件下载到宿主机
kubectl cp pod名:/容器路径 /本地路径

扩容 / 缩容、重启应用 ​

bash
# 副本扩缩容
kubectl scale deploy nginx --replicas=3

# 滚动重启deployment（优雅重启）
kubectl rollout restart deployment nginx

# 查看发布状态
kubectl rollout status deploy nginx
# 回滚上一个版本
kubectl rollout undo deploy nginx
# 查看历史版本
kubectl rollout history deploy nginx

命名空间 ​

bash
kubectl get ns                      # 查看所有命名空间
kubectl create ns test              # 创建命名空间
kubectl delete ns test              # 删除命名空间
kubectl get pods -n test            # 查指定ns资源
kubectl get pods --all-namespaces   # 查所有ns的pod

配置管理 ​

bash
# 创建cm
kubectl create cm config --from-file=app.conf
# 创建Secret
kubectl create secret generic mysecret --from-literal=user=admin
# 查看详情
kubectl get cm config -o yaml

端口转发调试 ​

bash
# 本地8080转发到pod的80端口
kubectl port-forward pod/xxx 8080:80
# 转发service
kubectl port-forward svc/nginx-svc 8080:80
