环境设置 ​

bash
# 系统版本OpenCloudOs9.6 由腾讯主导的系统 @see https://opencloudos.org/
# 更新组件
sudo dnf update -y

# 设置唯一主机名（集群每个节点不能重名）
sudo hostnamectl set-hostname k8s-master-1
sudo echo "192.168.0.151  k8s-master-1" | sudo tee -a /etc/hosts
# 放行端口
sudo firewall-cmd --permanent --add-port={6443,2379-2380,10250,10257,10259,30000-32767}/tcp
# 放行calico 网络插件 VXLAN 模式端口
sudo firewall-cmd --permanent --add-port=4789/udp
sudo firewall-cmd --reload

# 关闭 SELinux
sudo setenforce 0
sudo sed -i 's/^SELINUX=enforcing/SELINUX=disabled/' /etc/selinux/config

# 关闭swap （Kubernetes 的硬性要求）
swapoff -a
sudo sed -i '/swap/s/^/#/' /etc/fstab

# 时间同步工具,没有就安装
systemctl start chronyd
systemctl enable chronyd
# 验证时间同步
sudo chronyc sources
sudo chronyc makestep

# 临时加载
sudo modprobe overlay # 分层存储
sudo modprobe br_netfilter  # 桥接网络过滤

# 开机自动加载模块
sudo tee -a /etc/modules-load.d/k8s.conf << EOF
overlay
br_netfilter
EOF

# 内核参数优化（IP 转发、桥接流量）
sudo tee -a /etc/sysctl.d/k8s.conf << EOF
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF

sudo sysctl --system
# 验证
sysctl net.ipv4.ip_forward

放行网段 ​

  • 保证节点直接可以内网互通
bash
# 查看网段
ip route | grep src
# 集群节点之间全部内网互通（你的网段192.168.0.0/24）
sudo firewall-cmd --permanent --add-source=192.168.0.0/24 --zone=trusted

# 放行pod网段入站、出站
sudo firewall-cmd --permanent --add-source=10.244.0.0/16
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" source address="10.244.0.0/16" accept'

# 放行集群service网段（默认10.96.0.0/12）
sudo firewall-cmd --permanent --add-source=10.96.0.0/12
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" source address="10.96.0.0/12" accept'

# 重载生效
sudo firewall-cmd --reload

配置 containerd ​

  • 生成配置文件
bash
# 复制默认配置
sudo mkdir -p /etc/containerd
containerd config default | sudo tee /etc/containerd/config.toml
  • Cgroup 驱动改为 systemd（K8s 标准要求）
bash
sudo sed -i 's/SystemdCgroup \= false/SystemdCgroup \= true/g' /etc/containerd/config.toml
# 检查
cat /etc/containerd/config.toml | grep  SystemdCgroup
  • 配置沙盒版本
bash
sudo sed -i 's#sandbox_image = ".*"#sandbox_image = "registry.aliyuncs.com/google_containers/pause:3.10.1"#g' /etc/containerd/config.toml
cat /etc/containerd/config.toml | grep sandbox_image

镜像加速 ​

  • 参考下面结构创建目录
text
/etc/containerd/certs.d/
--| docker.io/ # docker仓库
--│     --| hosts.toml
--| harbor.test.com/  # 你的私有 Harbor 仓库配置
--|     --| hosts.toml
--|     --| ca.crt（自签证书可选）
bash
# 创建目录
sudo mkdir -p /etc/containerd/certs.d/
# docker镜像加速
sudo mkdir -p /etc/containerd/certs.d/docker.io
sudo tee /etc/containerd/certs.d/docker.io/hosts.toml >/dev/null <<EOF
server = "https://docker.io"

[host."https://docker.xuanyuan.me"]
capabilities = ["pull", "resolve"]
[host."https://docker.1ms.run"]
capabilities = ["pull", "resolve"]
[host."https://docker.m.daocloud.io"]
capabilities = ["pull", "resolve"]
[host."https://dockerproxy.com"]
capabilities = ["pull", "resolve"]
EOF

# 自定义镜像加速
sudo mkdir -p /etc/containerd/certs.d/harbor.test.com
sudo tee /etc/containerd/certs.d/harbor.test.com/hosts.toml >/dev/null <<EOF
server = "https://harbor.test.com"

[host."https://harbor.test.com"]
capabilities = ["pull", "resolve"]
# 跳过证书验证
skip_verify = true
EOF
  • 修改 /etc/containerd/config.toml
bash
sudo vim /etc/containerd/config.toml
# 找到 config_path
[plugins.'io.containerd.grpc.v1.cri'.registry]
config_path = "/etc/containerd/certs.d"

启动 containerd ​

bash
# 重新加载systemd并启动服务
sudo systemctl daemon-reload
sudo systemctl restart containerd
sudo systemctl enable containerd
# 验证服务状态
sudo systemctl status containerd

安装 crictl ​

  • k8s 操作containerd工具
bash
sudo dnf install -y cri-tools
# 生成配置
sudo tee -a /etc/crictl.yaml << EOF
runtime-endpoint: unix:///run/containerd/containerd.sock
image-endpoint: unix:///run/containerd/containerd.sock
timeout: 10
debug: false
EOF
# 检测
sudo crictl info
# 解决权限问题
sudo -E crictl ps -a

安装k8s ​

设置官方源 ​

bash
# 修改版本源
cat <<EOF | sudo tee /etc/yum.repos.d/kubernetes.repo
[kubernetes]
name=Kubernetes
baseurl=https://pkgs.k8s.io/core:/stable:/v1.35/rpm/
enabled=1
gpgcheck=1
gpgkey=https://pkgs.k8s.io/core:/stable:/v1.35/rpm/repodata/repomd.xml.key
EOF
# 清除缓存
sudo dnf clean all && sudo dnf makecache
# 核查版本
dnf list --showduplicates kubeadm kubelet kubectl

部署master节点 ​

清除残留 ​

bash
# 强制重置
sudo kubeadm reset -f
sudo rm -rf /etc/kubernetes /var/lib/kubelet /var/lib/etcd
# 重启容器运行时 & kubelet
sudo systemctl restart kubelet

初始化 ​

bash
sudo kubeadm init \
--kubernetes-version=1.35.5 \
--apiserver-advertise-address=192.168.0.151 \
--image-repository=registry.aliyuncs.com/google_containers \
--pod-network-cidr=10.244.0.0/16 \
--v=5
  • 参数说明
    • --kubernetes-version k8s版本
    • --apiserver-advertise-address apiserver监听地址 本机IP
    • --image-repository 镜像仓库
    • --pod-network-cidr Calico 默认网段

配置 kubectl 权限 ​

bash
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config

重新生成token ​

bash
kubeadm token create --print-join-command

网络插件 ​

  • 保证所有节点都能访问到网络插件镜像
  • 采用Calico网络插件
bash
# 下载配置，方便后续更改
wget https://docs.projectcalico.org/manifests/calico.yaml
# vim  calico.yaml 修改工作模式 VXLAN
# 去掉bgp
- name: CLUSTER_TYPE
    value: "k8s"
# 改为VXLAN
- name: CALICO_IPV4POOL_VXLAN
    value: "CrossSubnet"
# 关闭IPIP
- name: CALICO_IPV4POOL_IPIP
    value: "Never"
# 运行
kubectl apply -f calico.yaml
# 检查
kubectl get pods -n kube-system -w
