Prometheus ​

  • Prometheus 是开源时序数据库 (TSDB)+ 监控采集系统

核心特性 ​

  • 基于Pull 拉模式采集指标（服务自己暴露接口，Prometheus 定时拉取）
  • 本地存储时序数据，自带高效 TSDB，支持远程读写
  • 内置 PromQL 查询语言，灵活计算指标
  • 多维度标签（label）区分实例、服务、环境
  • 组件化架构，易扩展（采集、存储、告警、可视化解耦）
  • 适配容器 (K8s)、虚拟机、中间件、业务程序全场景监控

核心组件 ​

  • Prometheus Server：核心服务，抓取、存储、执行查询、触发告警规则
  • Exporter：各类服务指标采集适配器
    • 官方：node-exporter (服务器硬件)、mysql-exporter、redis-exporter、nginx-exporter
    • 自定义：Java/Python/Go 程序埋点暴露指标
  • AlertManager：告警聚合、去重、静默、路由分发
  • PushGateway：一次性任务、批处理程序推送指标

部署 ​

yaml
services:
    # node-exporter：采集宿主机CPU、内存、磁盘、网络指标
    node-exporter:
        image: prom/node-exporter:latest
        container_name: node-exporter
        restart: always
        volumes:
            - /proc:/host/proc:ro
            - /sys:/host/sys:ro
            - /:/rootfs:ro
        command:
            - '--path.procfs=/host/proc'
            - '--path.sysfs=/host/sys'
            - '--collector.filesystem.mount-points-exclude=^/(sys|proc|dev|host|etc)($|/)'
        ports:
            - '9100:9100'

    # AlertManager 告警管理器
    alertmanager:
        image: prom/alertmanager:latest
        container_name: alertmanager
        restart: always
        volumes:
            - ./volumes/alertmanager/alertmanager.yml:/etc/alertmanager/alertmanager.yml
            - ./volumes/alertmanager/data:/alertmanager
        ports:
            - '9093:9093'

    # Prometheus 服务
    prometheus:
        image: prom/prometheus:latest
        container_name: prometheus
        restart: always
        volumes:
            - ./volumes/prometheus/prometheus.yml:/etc/prometheus/prometheus.yml
            - ./volumes/prometheus/rules:/etc/prometheus/rules
            - ./volumes/prometheus/data:/prometheus # 数据持久化
        command:
            - '--config.file=/etc/prometheus/prometheus.yml'
            - '--storage.tsdb.path=/prometheus'
            - '--web.enable-lifecycle' # 支持热重载配置 curl -X POST http://ip:9090/-/reload
        ports:
            - '9090:9090'
        depends_on:
            - node-exporter
            - alertmanager

    - 'rules/*.yml'

          - targets: ['node-exporter:9100'] # ip:端口 / 容器名称：端口 / 宿主机：端口

alert.yml ​

  • 位置：./volumes/prometheus/rules/alert.yml, 没有就创建
yml
groups:
    - name: 'node-alert'
      rules:
          # CPU使用率大于80%告警
          - alert: CPUHighUsage
            expr: 100 - (avg by (instance) (rate(node_cpu_seconds_total{mode="idle"}[5m])) * 100) > 80
            for: 2m
            labels:
                severity: warning
            annotations:
                summary: '服务器CPU占用过高于80%'
                description: '实例 {{ $labels.instance }} CPU使用率: {{ $value }}%'

          # 内存使用率大于80%告警
          - alert: MemoryHighUsage
            expr: (1 - (node_memory_MemFree_bytes + node_memory_Buffers_bytes + node_memory_Cached_bytes + node_memory_SReclaimable_bytes) / node_memory_MemTotal_bytes) * 100 > 80
            for: 2m
            labels:
                severity: warning
            annotations:
                summary: '服务器内存占用高于80%'
                description: '实例 {{ $labels.instance }} 内存使用率: {{ $value }}%'

          # 磁盘根分区使用率超过80%
          - alert: DiskHighUsage
            expr: (1 - node_filesystem_avail_bytes{mountpoint="/",fstype!~"tmpfs|devtmpfs"} / node_filesystem_size_bytes{mountpoint="/",fstype!~"tmpfs|devtmpfs"}) * 100 > 80
            for: 2m
            labels:
                severity: critical
            annotations:
                summary: '根磁盘空间占满超过80%'
                description: '实例 {{ $labels.instance }} 根分区使用率: {{ $value }}%'

          # 节点宕机（target状态为0代表失联）
          - alert: NodeDown
            expr: up == 0
            for: 1m
            labels:
                severity: critical
            annotations:
                summary: '节点 {{ $labels.instance }} 失联'
                description: '采集任务 {{ $labels.job }} 实例 {{ $labels.instance }} 心跳中断'

alertmanager.yml ​

  • 位置：./volumes/alertmanager/alertmanager.yml,没有就创建
yml
# 全局配置
global:
    resolve_timeout: 5m
    # 配置邮件通知
    smtp_from: 'xxx@qq.com'
    smtp_smarthost: 'smtp.qq.com:465'
    smtp_auth_username: 'xxx@qq.com'
    smtp_auth_password: 'QQ邮箱第三方授权码'
    smtp_require_tls: false

# 告警路由根节点
route:
    group_by: ['alertname', 'instance'] # 按告警名+实例分组告警
    group_wait: 10s # 第一次告警触发后，等待10s聚合同组告警再发送
    group_interval: 2m # 一组告警已发过后，2分钟内新增告警不再推送
    repeat_interval: 2h # 告警未恢复，每2小时重复推送一次告警
    receiver: 'default-email' # 默认接收人

# 所有接收者配置
receivers:
    # 邮件接收
    - name: 'default-email'
      email_configs:
          - to: 'admin@xxx.com'
            send_resolved: true # 告警恢复时发送通知
    # 钉钉 严重告警
    - name: 'dingtalk-critical'
      webhook_configs:
          - url: 'http://127.0.0.1:8060/dingtalk/critical/send'
            send_resolved: true # 告警恢复时发送通知
    # 钉钉 普通警告
    - name: 'dingtalk-warning'
      webhook_configs:
          - url: 'http://127.0.0.1:8060/dingtalk/warn/send'
            send_resolved: true # 告警恢复时发送通知

# 告警抑制规则（核心：节点宕机，抑制该机器所有服务告警）
inhibit_rules:
    - source_matchers:
          - alertname = NodeDown
      target_matchers:
          - instance =~ .+
      equal: ['instance']
