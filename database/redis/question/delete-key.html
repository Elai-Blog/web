    如何批量删除指定前缀的key?

页面导航

Redis如何批量删除指定前缀的key? ​

  • 问题核心：由于Redis是单线程的，所以在执行keys命令时，会阻塞其他命令的执行。当keys量非常大时，会造成阻塞，导致redis其他服务不可用。

批量删除 ​

  • keys pattern命令，获取相关key 然后全部遍历删除
  • 优点：一次精确删除
  • 缺点：keys量非常大时，会造成阻塞，导致redis其他服务不可用。
go

func main() {
	ctx := context.Background()
	rds := GetRedis()

    // 先插入 100 数据
    for i := 0; i < 100; i++ {
        key := fmt.Sprintf("user:%d", i)
        fmt.Println("开始插入:", key)
        rds.Set(ctx, key, 1, time.Hour)
    }

    // 使用keys 命令获取所有key
    keys, err := rds.Keys(ctx, "user:*").Result()

方案二 ​

  • SCAN分批匹配，然后删除。

  • 优点：分批执行，不会造成阻塞。

  • 缺点：

    • 需要分多次执行，花费时间长。扫描结果可能存在重复情况，需要客户端去重。
    • count值不能太小，总数的10%左右比较合适。
go
func main() {
	ctx := context.Background()
	rds := GetRedis()
	// 先插入 10w  数据
	for i := 0; i < 1000; i++ {
		key := fmt.Sprintf("user:%d", i)
		fmt.Println("开始插入:", key)
		rds.Set(ctx, key, 1, time.Hour)
	}
	fmt.Println("开始删除")
	var cursor uint64 = 0
	var count int64 = 100 // 每次约为总数的10%
	var total int64 = 0
	for {
		keys, newCursor := rds.Scan(ctx, cursor, "user:*", count).Val()
		deleteNum, err := rds.Del(ctx, keys...).Result()
		if err != nil {
			fmt.Println("删除失败！", err)
			break
		}
		total += deleteNum
		fmt.Println("删除了:", deleteNum, "个数据")
		// 游标为 0 表示迭代结束
		if newCursor == 0 {
			break
		}
		cursor = newCursor
	}
	fmt.Println("删除完成，共删除了:", total, "个数据")
}
