地理位置 ​

  • 主要用于存储地理位置信息，并对存储的信息进行操作

操作命令 ​

命令功能
GEOADD添加一个或多个地理空间位置（经度、纬度、成员名称）到指定的键中
GEOPOS返回一个或多个成员的经度和纬度
GEODIST返回两个成员之间的距离
GEORADIUS返回指定半径内的成员
GEORADIUSBYMEMBER返回指定成员周围的成员
ZREM从有序集合中删除一个或多个成员

应用场景 ​

计算距离 ​

go
func main() {
	ctx := context.Background()
	rds := GetRedis()

	// 添加一组地理位置
	rds.GeoAdd(ctx, "city", &redis.GeoLocation{
		Name:      "shanghai",
		Longitude: 121.473701,
		Latitude:  31.230378,
	}, &redis.GeoLocation{
		Name:      "beijing",
		Longitude: 116.407396,
		Latitude:  39.904989,
	})

	// 获取地理位置
	shanghai := rds.GeoPos(ctx, "city", "shanghai").Val()
	beijing := rds.GeoPos(ctx, "city", "beijing").Val()
	fmt.Println(shanghai[0], beijing[0])

	// 计算两个地理位置之间的距离
	jl := rds.GeoDist(ctx, "city", "shanghai", "beijing", "km").Val()
	fmt.Println("北京-上海的距离", jl)
}

附近的人 ​

go
func main() {
	ctx := context.Background()
	rds := GetRedis()

	long := 116.407396 // 上海-经度
	lat := 39.904989   // 上海-纬度
	for i := 0; i < 100; i++ {
		if rand.Intn(100) < 30 {
			continue
		}
		// 以上海为基准 随机添加一些位置的人
		xlong := long + rand.Float64()*0.01
		ylat := lat + rand.Float64()*0.01
		name := fmt.Sprintf("member_%d", i)
		rds.GeoAdd(ctx, "member", &redis.GeoLocation{
			Name:      name,
			Longitude: xlong,
			Latitude:  ylat,
		})
	}
	ls := rds.GeoRadius(ctx, "member", long, lat, &redis.GeoRadiusQuery{
		Count:       10,
		Radius:      3,
		Unit:        "km",
		Sort:        "ASC", // 按距离升序排列（近到远）
		WithDist:    true,  // 强制返回距离（Dist 字段）
		WithCoord:   true,  // 强制返回坐标（Coord 字段）
		WithGeoHash: true,  // 强制返回 GeoHash（验证数据有效性）
	}).Val()
	for _, v := range ls {
		fmt.Printf("成员：%s,距离：%.2f km,经度：%.2f,纬度：%.2f\n", v.Name, v.Dist, v.Longitude, v.Latitude)
	}
}
