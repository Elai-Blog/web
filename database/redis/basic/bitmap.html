位图（Bitmap） ​

  • Redis Bitmap（位图）是一种基于字符串数据结构实现的高效存储方案，本质是将字符串中的每个字节拆分为 8 个二进制位（bit），通过对单个 bit 的读写来实现数据存储与计算。

  • 它的核心优势是极致的空间效率——1 个字节可存储 8 个布尔型（是 / 否）数据，1MB 可存储约 838 万条状态，1GB 可存储近 85 亿条状态，非常适合处理海量低维度数据（如用户在线状态、签到记录、权限标记等）。

操作命令 ​

命令说明
SETBIT key offset value设置 key 中指定偏移量 offset 的 bit 值为 value（0 或 1）
GETBIT key offset获取 key 中指定偏移量 offset 的 bit 值
BITCOUNT key [start end]统计 key 中指定范围 [start, end] 内的 bit 值为 1 的数量
BITOP operation destkey key [key ...]对一个或多个 key 进行位操作（AND、OR、XOR、NOT），将结果存储到 destkey 中
BITPOS key bit [start end]查找 key 中指定范围 [start, end] 内第一个值为 bit 的偏移量

应用场景 ​

在线状态 ​

  • 注意事项：
    • 位图的每个 bit 只能存储 0 或 1，无法存储其他值。
    • 用户下线只能手动触发，不能依赖 Redis 自动过期。
go
func main() {
	ctx := context.Background()
	rds := GetRedis()

	// 记录用户在线
	for i := 0; i < 100; i++ {
		userId := 10000 + int64(i)
		if rand.Intn(100) < 30 {
			// 随机一些不在线的用户
			rds.SetBit(ctx, "online:user", userId, 0)
			continue
		}
		// 设置一些在线用户
		rds.SetBit(ctx, "online:user", userId, 1)
		// 设置有效期5分钟
		key := "online:user:" + strconv.FormatInt(userId, 10)
		rds.Set(ctx, key, 1, 5*time.Minute)
	}

	// 统计在线人数
	count := rds.BitCount(ctx, "online:user", &redis.BitCount{
		Start: 0,
		End:   -1,
	}).Val()
	fmt.Println("在线人数统计：", count)

	// 周期性检查离线用户
	for {
		removeOffline(ctx, rds)
		time.Sleep(10 * time.Second)
	}
}

// 删除离线的用户
func removeOffline(ctx context.Context, rdb *redis.Client) {
	key := "online:user"
	// 获取位图的字节数组
	data, err := rdb.Get(ctx, key).Bytes()
	if err != nil {
		fmt.Printf("获取位图失败: %v\n", err)
		return
	}
	for i := 0; i < len(data); i++ {
		// 获取当前字节的二进制表示
		byteData := fmt.Sprintf("%08b", data[i])
		// 遍历字节的每一位
		for j := 0; j < 8; j++ {
			bit := byteData[j]
			// 判断当前位是否为1
			if bit == '1' {
				userId := i*8 + j
				expireKey := "online:user:" + fmt.Sprintf("%d", userId)
				// 如果当前用户的有效期已过，则将其从在线列表中删除
				if rdb.Exists(ctx, expireKey).Val() == 0 {
					// 标记离线
					rdb.SetBit(ctx, key, int64(userId), 0)
				}
			}
		}
	}
}
